    ```python
    """Report Source-Schema Auditor (Phase 03.4 Plan 05). Per CONTEXT D-10.

    For every Report in university_erp scope (Education filtered per D-02), verify:
      - source DocType field names exist in meta
      - filter field names exist
      - join field names exist (Report Builder / Query Report inspection)
      - group-by columns exist

    Out of scope (per D-10 carryover): ERPNext/HRMS report failures due to missing
    Company setup. These are skipped (filtered by module).
    """
    import os
    import re
    import json
    import frappe

    UNIVERSITY_MODULES = [
        "University ERP", "University Academics", "University Admissions",
        "University Student Info", "University Examinations", "University Finance",
        "Faculty Management", "University Hostel", "University Transport",
        "University Library", "University Placement", "University LMS",
        "University Research", "University OBE", "University Integrations",
        "University Portals", "University Payments", "University Inventory",
        "University Grievance", "University Analytics", "University Feedback",
    ]

    META_FIELDS_BUILTIN = {"name", "owner", "creation", "modified", "modified_by", "docstatus", "idx", "parent", "parenttype", "parentfield"}

    def _find_report_py(report_name):
        """Find <report>.py for a Query/Script report by name slug."""
        slug = frappe.scrub(report_name)
        for app_path in frappe.get_installed_apps():
            base = frappe.get_app_path(app_path)
            for root, dirs, files in os.walk(base):
                if slug + ".py" in files and "/report/" in root.replace("\\", "/"):
                    return os.path.join(root, slug + ".py")
        return None

    def audit_report_source_schema(report_name):
        """Per D-10: verify every fieldname referenced by a report exists on its source DocType."""
        rpt = frappe.get_doc("Report", report_name)
        source_dt = rpt.ref_doctype
        if not source_dt:
            return {"report": report_name, "status": "SKIP", "reason": "no ref_doctype"}
        if not frappe.db.exists("DocType", source_dt):
            return {"report": report_name, "status": "FAIL", "issues": [f"ref_doctype '{source_dt}' missing"]}

        meta = frappe.get_meta(source_dt)
        valid_fields = {f.fieldname for f in meta.fields} | META_FIELDS_BUILTIN
        # Include child-table fields (joins via parent doctype)
        for cf in meta.get_table_fields():
            try:
                cmeta = frappe.get_meta(cf.options)
                valid_fields |= {f.fieldname for f in cmeta.fields}
            except Exception:
                pass

        issues = []

        # 1. Report Builder — columns & filters in stored JSON
        if rpt.report_type == "Report Builder":
            try:
                cols = json.loads(rpt.json or "{}")
                for col in (cols.get("columns") or []):
                    fn = col[0] if isinstance(col, list) else col
                    if fn and fn not in valid_fields:
                        issues.append(f"report-builder column fieldname '{fn}' not on {source_dt}")
                for filt in (cols.get("filters") or []):
                    fn = filt[1] if isinstance(filt, list) and len(filt) > 1 else None
                    if fn and fn not in valid_fields:
                        issues.append(f"report-builder filter fieldname '{fn}' not on {source_dt}")
            except (TypeError, json.JSONDecodeError) as e:
                issues.append(f"failed to parse report.json: {e}")

        # 2. Query Report / Script Report — parse the .py for fieldname references
        elif rpt.report_type in ("Query Report", "Script Report"):
            py_file = _find_report_py(report_name)
            if py_file and os.path.exists(py_file):
                with open(py_file) as f:
                    src = f.read()
                # Find fieldname patterns
                referenced = set(re.findall(r'["\']fieldname["\']\s*:\s*["\'](\w+)["\']', src))
                # Find SQL column refs (best-effort, alias-prefixed columns parsed)
                sql_cols = set(re.findall(r'(?:tab[A-Z]\w+\.)?`?(\w+)`?\s*(?:,|\bFROM\b|\bWHERE\b|\bAS\b)', src, re.IGNORECASE))
                # Filter to fieldnames that look like DocType fields (lowercase_underscore)
                referenced |= {c for c in sql_cols if c.islower() or "_" in c}
                for fn in referenced:
                    if fn in valid_fields or fn in {"COUNT", "SUM", "AVG", "MIN", "MAX", "IF", "IFNULL", "CASE", "WHEN", "THEN", "ELSE", "END"}:
                        continue
                    if len(fn) <= 2 or fn.isupper():
                        continue  # likely SQL keyword/alias
                    # Best-effort heuristic — may have false positives; flag as WARN not FAIL
                    issues.append(f"WARN unknown fieldname '{fn}' in {os.path.basename(py_file)}")

        return {"report": report_name, "source_dt": source_dt, "status": "PASS" if not [i for i in issues if not i.startswith("WARN")] else "FAIL",
                "issues": issues}

    def discover_reports():
        return frappe.get_all(
            "Report",
            filters={"module": ["in", UNIVERSITY_MODULES]},
            fields=["name", "ref_doctype", "report_type", "module"],
            order_by="module, name",
        )

    def run():
        results = []
        for r in discover_reports():
            try:
                results.append(audit_report_source_schema(r["name"]))
            except Exception as e:
                results.append({"report": r["name"], "status": "ERROR", "issues": [str(e)[:200]]})

        # Write consolidated AUDIT_REPORT.md per D-12
        report_path = os.path.normpath(os.path.join(
            frappe.get_app_path("university_erp"), "..", "..", "..", "..",
            ".planning", "phases", "03.4-post-fork-backend-audit-fix",
            "03.4-05-AUDIT_REPORT.md",
        ))
        passed = sum(1 for r in results if r["status"] == "PASS")
        failed = sum(1 for r in results if r["status"] == "FAIL")
        warned = sum(1 for r in results if any(i.startswith("WARN") for i in r.get("issues", [])))
        with open(report_path, "w") as f:
            f.write("# Phase 03.4 Plan 05 — Report Source-Schema Audit (D-10)\n\n")
            f.write(f"**Source filter:** module ∈ UNIVERSITY_MODULES (Education filtered per D-02)\n\n")
            f.write(f"**Total reports:** {len(results)} | **PASS:** {passed} | **FAIL:** {failed} | **WARN:** {warned}\n\n")
            f.write("## Per-Report Findings\n\n")
            f.write("| Report | Source DocType | Status | Issues |\n")
            f.write("|--------|----------------|--------|--------|\n")
            for r in results:
                src = r.get("source_dt", "")
                issues = "; ".join(r.get("issues", [])) or "—"
                f.write(f"| {r['report']} | {src} | {r['status']} | {issues} |\n")
            f.write(f"\n## Out of Scope (per D-10)\n\n")
            f.write("- ERPNext/HRMS report failures due to missing Company setup are skipped (filtered by module).\n\n")
            f.write(f"## Status: {'PASS' if failed == 0 else 'FAIL'}\n")
        print(f"Audited {len(results)} reports → {report_path}")
        return {"total": len(results), "passed": passed, "failed": failed, "warned": warned}
    ```
