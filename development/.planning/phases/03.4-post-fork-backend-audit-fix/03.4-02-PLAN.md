    ```python
    """Workflow E2E auditor (Phase 03.4 Plan 02). Per CONTEXT D-04, D-05.

    Discovers all active workflows in tabWorkflow (filtered to university modules),
    walks each one happy-path via apply_workflow, verifies final state, deletes test doc.
    """
    import json
    import os
    import frappe
    from frappe.model.workflow import apply_workflow, get_transitions

    UNIVERSITY_MODULES = [
        "University ERP", "University Academics", "University Admissions",
        "University Student Info", "University Examinations", "University Finance",
        "Faculty Management", "University Hostel", "University Transport",
        "University Library", "University Placement", "University LMS",
        "University Research", "University OBE", "University Integrations",
        "University Portals", "University Payments", "University Inventory",
        "University Grievance", "University Analytics", "University Feedback",
    ]

    def _create_test_doc_for(doctype):
        """Build minimal test doc using demo-data FK targets where required."""
        # Per workflow: Hostel Allocation needs Student + Hostel Room (use demo seed);
        # Leave Application needs Employee; etc. Returns inserted doc.
        meta = frappe.get_meta(doctype)
        defaults = {"doctype": doctype}
        # Auto-fill required Link fields from demo data
        for f in meta.fields:
            if f.reqd and f.fieldtype == "Link" and f.options:
                target = frappe.db.get_value(f.options, {}, "name")
                if target:
                    defaults[f.fieldname] = target
        doc = frappe.get_doc(defaults)
        doc.flags.ignore_mandatory = True
        doc.flags.ignore_validate = True
        doc.insert(ignore_permissions=True)
        return doc

    def walk_all_transitions(workflow_name):
        """Walk a single workflow happy-path. Returns {workflow, status, visited, transitions, error?}."""
        wf = frappe.get_doc("Workflow", workflow_name)
        doctype = wf.document_type
        state_field = wf.workflow_state_field or "workflow_state"
        frappe.set_user("Administrator")
        try:
            doc = _create_test_doc_for(doctype)
        except Exception as e:
            return {"workflow": workflow_name, "doctype": doctype, "status": "FAIL",
                    "visited": [], "transitions": [], "error": f"factory: {str(e)[:200]}"}

        visited = [doc.get(state_field)]
        transitions_log = []
        try:
            for _ in range(20):  # safety bound
                transitions = get_transitions(doc)
                if not transitions:
                    break
                action = transitions[0]["action"]
                apply_workflow(doc, action)
                visited.append(doc.get(state_field))
                transitions_log.append(action)
            return {"workflow": workflow_name, "doctype": doctype, "status": "PASS",
                    "visited": visited, "transitions": transitions_log}
        except Exception as e:
            return {"workflow": workflow_name, "doctype": doctype, "status": "FAIL",
                    "visited": visited, "transitions": transitions_log, "error": str(e)[:300]}
        finally:
            try:
                frappe.delete_doc(doctype, doc.name, force=1, ignore_permissions=True)
            except Exception:
                pass
            frappe.db.commit()

    def discover_workflows():
        """List active workflows whose document_type is in a university_erp or university_finance module."""
        all_wf = frappe.get_all("Workflow", filters={"is_active": 1}, fields=["name", "document_type"])
        result = []
        for wf in all_wf:
            mod = frappe.db.get_value("DocType", wf["document_type"], "module")
            if mod in UNIVERSITY_MODULES:
                result.append({"name": wf["name"], "document_type": wf["document_type"], "module": mod})
        return result

    def run():
        """Entry point: bench --site <site> execute university_erp.scripts.audit_workflows_e2e.run"""
        results = []
        workflows = discover_workflows()
        print(f"Discovered {len(workflows)} workflows in university_erp scope")
        for wf in workflows:
            r = walk_all_transitions(wf["name"])
            results.append(r)
            print(f"  [{r['status']}] {wf['name']} ({wf['document_type']}) — {len(r['transitions'])} transitions")

        # Write consolidated AUDIT_REPORT.md
        report_path = os.path.join(
            frappe.get_app_path("university_erp"), "..", "..", "..", "..",
            ".planning", "phases", "03.4-post-fork-backend-audit-fix",
            "03.4-02-AUDIT_REPORT.md",
        )
        report_path = os.path.normpath(report_path)
        passed = sum(1 for r in results if r["status"] == "PASS")
        failed = sum(1 for r in results if r["status"] == "FAIL")
        with open(report_path, "w") as f:
            f.write(f"# Phase 03.4 Plan 02 — Workflow E2E Audit\n\n")
            f.write(f"**Total workflows in univ scope:** {len(results)}\n")
            f.write(f"**PASS:** {passed} | **FAIL:** {failed}\n\n")
            f.write("## Per-Workflow Results\n\n")
            f.write("| Workflow | DocType | Status | Visited States | Transitions | Error |\n")
            f.write("|----------|---------|--------|----------------|-------------|-------|\n")
            for r in results:
                visited = " → ".join(str(v) for v in r["visited"])
                trans = ", ".join(r["transitions"])
                err = r.get("error", "")
                f.write(f"| {r['workflow']} | {r['doctype']} | {r['status']} | {visited} | {trans} | {err} |\n")
            f.write(f"\n## Status: {'PASS' if failed == 0 else 'FAIL'}\n")
        print(f"Report written to {report_path}")
        return {"total": len(results), "passed": passed, "failed": failed}
    ```
