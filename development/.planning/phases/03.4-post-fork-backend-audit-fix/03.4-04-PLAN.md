    Append the following function to `audit_cross_app_links.py` and wire it into the existing audit flow. Verify the file already has `UNIVERSITY_MODULES` constant or define at top.

    ```python
    UNIVERSITY_MODULES = [
        "University ERP", "University Academics", "University Admissions",
        "University Student Info", "University Examinations", "University Finance",
        "Faculty Management", "University Hostel", "University Transport",
        "University Library", "University Placement", "University LMS",
        "University Research", "University OBE", "University Integrations",
        "University Portals", "University Payments", "University Inventory",
        "University Grievance", "University Analytics", "University Feedback",
    ]

    def _validate_d03(target_doctype, dt_to_module=None):
        """Per CONTEXT D-03: two-condition check.
        (a) Target DocType exists in DB.
        (b) Target's module is in UNIVERSITY_MODULES (or legitimate cross-app).
        Returns three-state: PASS, EXTERNAL_OK, BROKEN with reason.
        """
        # Condition (a)
        if not frappe.db.exists("DocType", target_doctype):
            return ("BROKEN", f"Target DocType '{target_doctype}' does not exist")
        # Condition (b)
        target_module = (dt_to_module or {}).get(target_doctype) or frappe.db.get_value(
            "DocType", target_doctype, "module"
        )
        if target_module in UNIVERSITY_MODULES:
            return ("PASS", "")
        if target_module == "Education":
            return ("BROKEN", f"Target '{target_doctype}' owned by deprecated Education module")
        # ERPNext/HRMS/Frappe core etc. — legitimate cross-app
        return ("EXTERNAL_OK", f"Target '{target_doctype}' in non-univ module '{target_module}'")
    ```

    Then patch the main audit loop. Find the existing iteration over Link/Dynamic Link/Table fields (likely inside `_test_link_integrity` or run()) and replace per-field processing with:

    ```python
    findings.append({
        "source_doctype": source_dt,
        "fieldname": fieldname,
        "fieldtype": fieldtype,  # Link / Dynamic Link / Table
        "target_doctype": target_dt,
        "status": status,
        "reason": reason,
    })
    ```
