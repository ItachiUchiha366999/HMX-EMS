    ```python
    import json

    def _audit_workspace_consistency():
        """Per CONTEXT D-06: For every public Workspace, parse content JSON,
        extract card_name values; cross-reference vs tabWorkspace Link rows
        where type='Card Break'. FAIL if any Card Break in `links` is missing
        from `content`. Catches the 03.3.1 failure mode (39 forked doctypes
        invisible in Desk).
        """
        findings = []
        workspaces = frappe.get_all(
            "Workspace",
            filters={"public": 1, "module": ["in", UNIVERSITY_MODULES]},  # D-02
            fields=["name", "module", "title", "content"],
        )
        for ws in workspaces:
            try:
                content = json.loads(ws["content"] or "[]")
            except (TypeError, json.JSONDecodeError):
                findings.append({
                    "workspace": ws["name"], "module": ws["module"],
                    "status": "FAIL", "issue": "Invalid content JSON",
                })
                continue

            content_cards = {
                block.get("data", {}).get("card_name")
                for block in content
                if isinstance(block, dict) and block.get("type") == "card"
            }
            content_cards.discard(None)

            link_card_labels = set(
                row["label"]
                for row in frappe.get_all(
                    "Workspace Link",
                    filters={"parent": ws["name"], "type": "Card Break"},
                    fields=["label"],
                )
            )

            # Pitfall 1: also try frappe.scrub for label↔card_name normalization
            link_cards_normalized = link_card_labels | {frappe.scrub(l) for l in link_card_labels}

            missing_in_content = link_card_labels - content_cards
            # Re-check with scrubbed
            missing_after_scrub = {l for l in missing_in_content if frappe.scrub(l) not in content_cards}

            orphan_in_content = content_cards - link_cards_normalized

            if missing_after_scrub or orphan_in_content:
                findings.append({
                    "workspace": ws["name"], "module": ws["module"],
                    "status": "FAIL",
                    "missing_in_content": sorted(missing_after_scrub),
                    "orphan_in_content": sorted(orphan_in_content),
                })
            else:
                findings.append({
                    "workspace": ws["name"], "module": ws["module"],
                    "status": "PASS",
                })
        return findings
    ```

       ```python
       """Restrict ERPNext Accounts original reports to System Manager only (D-11).

       University users see the university_finance versions (Fee Receivable,
       Vendor Payable, Student Ledger Summary). ERPNext originals remain visible
       only to System Manager. Idempotent — re-runnable.
       """
       import frappe

       ERPNEXT_ACCOUNTS_REPORTS = [
           "Accounts Receivable", "Accounts Payable", "General Ledger",
           "Sales Register", "Purchase Register", "Trial Balance",
           "Profit and Loss Statement", "Balance Sheet", "Cash Flow",
           "Item-wise Sales Register", "Item-wise Purchase Register",
           "Bank Reconciliation Statement", "Customer Ledger Summary",
           "Supplier Ledger Summary", "TDS Computation Summary",
           "Sales Order Analysis", "Purchase Order Analysis",
       ]

       def run():
           restricted = 0
           for rname in ERPNEXT_ACCOUNTS_REPORTS:
               if not frappe.db.exists("Report", rname):
                   print(f"  [SKIP] {rname} not installed")
                   continue
               # Check if already restricted to System Manager only
               roles = frappe.get_all("Has Role",
                                      filters={"parent": rname, "parenttype": "Report"},
                                      pluck="role")
               if roles == ["System Manager"]:
                   print(f"  [SKIP] {rname} already restricted")
                   continue
               # Apply restriction
               rpt = frappe.get_doc("Report", rname)
               rpt.set("roles", [{"role": "System Manager"}])
               rpt.save(ignore_permissions=True)
               restricted += 1
               print(f"  [RESTRICTED] {rname}")
           frappe.db.commit()
           return {"restricted": restricted}
       ```
